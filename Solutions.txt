# Write a program to count number of vowels.
word=input("Enter a string:")
words=word.upper()
count=0
for i in range(len(word)):
    if words[i]=="A" or words[i]=="E" or words[i]=="I" or words[i]=="O" or words[i]=="U":
        count+=1
print(count)

# Write a program to count number of consonents.
words=input("Enter a string:")
words=words.upper()
count=0
for i in range(len(words)):
    if words[i]!="A" and words[i]!="E" and words[i]!="I" and words[i]!="O" and words[i]!="U":
        count+=1
print(count)

# Write a program to reverse a string without using slicing.
