# Write a program to count uppercase and lowercase letters in a string.
w=input("Enter the string")
count_lower=0
count_upper=0
for i in range(len(w)):
    if w[i].isupper():
        count_upper+=1
    elif w[i].islower():
        count_lower+=1
print(count_lower,count_upper)

# Write a program to count uppercase and lowercase letters in a string without using isupper and islower.
s=input("Enter the string")
count_lower=0
count_upper=0
for i in range(len(s)):
    if ord(s[i])>=97 and ord(s[i])<=122:
        count_lower+=1
    elif ord(s[i])>=65 and ord(s[i])<=90:
        count_upper+=1
print(count_lower,count_upper)

# Write a program to convert a string into uppercase using ASCII calues.
s=input("Enter the string")
new=""
for i in range(len(s)):
    if ord(s[i])>=97 and ord(s[i])<=122:
        new=new+chr(ord(s[i])-32)
    elif ord(s[i])>=65 and ord(s[i])<=90:
        new=new+s[i]
print(new)
